lastOtherSport : '', qty: qty }); }); if (!items.length) { throw new Error('No names were entered.'); } if (totalQty !== purchasedQty) { throw new Error( 'Entered quantity (' + totalQty + ') does not match Etsy purchased quantity (' + purchasedQty + ').' ); } return { orderNumber: orderNumber, purchasedQty: purchasedQty, items: items, totalQty: totalQty }; } function writeOrderBlock_(sheet, order, requestId, submissionKey) { // One blank line between orders keeps the sheet easy to read. sheet.appendRow(['']); sheet.appendRow([ 'Etsy Order Number', order.orderNumber, 'Submitted At', new Date(), 'Request ID', requestId, 'Submission Key', submissionKey ]); sheet.appendRow([ 'Total Quantity Purchased', order.purchasedQty, 'Names Entered', order.items.length ]); sheet.appendRow(['No.', 'Name', 'Color', 'Font', 'Sport', 'Qty']); const rows = order.items.map(function(item) { let sportDisplay = item.sport || 'None'; if (item.sport === 'Other' && item.otherSport) { sportDisplay = 'Other: ' + item.otherSport; } return [ item.no, item.name, item.color, item.font, sportDisplay, item.qty ]; }); const startRow = sheet.getLastRow() + 1; sheet.getRange(startRow, 1, rows.length, 6).setValues(rows); } function makeSubmissionKey_(order) { const canonical = JSON.stringify({ orderNumber: order.orderNumber, purchasedQty: order.purchasedQty, items: order.items.map(function(item) { return { name: item.name, color: item.color, font: item.font, sport: item.sport, otherSport: item.otherSport, qty: item.qty }; }) }); const digest = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, canonical, Utilities.Charset.UTF_8 ); return digest.map(function(byte) { const value = byte < 0 ?
